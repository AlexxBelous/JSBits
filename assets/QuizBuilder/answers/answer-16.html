const list = document.getElementById("list");
const originalItems = Array.from(list.children);

function limitDOMList(n) {
    const counts = new Map();

    originalItems.forEach(item => {
    // Получаем идентификатор места
        const id = item.dataset.id;
    // Сколько раз уже встречался
        const count = counts.get(id) || 0;

        if (count < n) {
            counts.set(id, count + 1);
    // показываем элемент
            item.classList.remove("hidden");
        } else {
     // скрываем лишний
            item.classList.add("hidden");
        }
    });
}

document.getElementById("apply").addEventListener("click", () => {
    // Читаем лимит из input
    const n = Number(document.getElementById("limit").value);
    // Применяем фильтр к списку
    limitDOMList(n);
});
