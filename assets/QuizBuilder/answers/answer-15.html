function solve(arr) {
    const MINUTES_IN_DAY = 1440;

     // 1. Убираем дубликаты
    const uniqueTimes = [...new Set(arr)];

    // 2. Переводим в минуты
    const minutes = uniqueTimes.map(time => {
        const [h, m] = time.split(":").map(Number);
        return h * 60 + m;
    });

    // 3. Сортируем
    minutes.sort((a, b) => a - b);

    let maxGap = 0;

    // 4. Разницы между соседними будильниками
    for (let i = 0; i < minutes.length - 1; i++) {
        const gap = minutes[i + 1] - minutes[i] - 1;
        maxGap = Math.max(maxGap, gap);
    }

    // 5. Разница через полночь
    const nightGap = (MINUTES_IN_DAY - minutes[minutes.length - 1] + minutes[0]) - 1;
    maxGap = Math.max(maxGap, nightGap);

    // 6. Форматируем ответ в HH:MM
    const hours = String(Math.floor(maxGap / 60)).padStart(2, "0");
    const mins = String(maxGap % 60).padStart(2, "0");

    return `${hours}:${mins}`;
}

    // Примеры
solve(["14:51"])
    // → "23:59"


solve(["23:00","04:22","18:05","06:24"])
    // → "11:40"
