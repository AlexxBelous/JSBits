// 1. Получаем элементы из DOM
const list = document.getElementById("list");
const checkBtn = document.getElementById("checkBtn");

// 2. Создаем функцию для проверки последовательности
function runCheck() {
    // 2.1. Очищаем предыдущий список
    list.innerHTML = "";

    // 2.2. Получаем случайный массив чисел
    const numbers = randomNumber();

    // 2.3. Создаем элементы списка для каждого числа
    numbers.forEach(num => {
        const li = document.createElement("li");
        li.textContent = num;
        list.appendChild(li);
    });

    // 2.4. Получаем все элементы li для дальнейшей проверки
    const items = Array.from(list.children);

    // 2.5. Проверяем, идут ли числа подряд
    for (let i = 1; i < numbers.length; i++) {
        // Если текущее число не на 1 больше предыдущего — подсвечиваем
        if (numbers[i] !== numbers[i-1] + 1) {
            items[i].classList.add("unsorted");
        }
    }
}

// 3. Назначаем обработчик клика на кнопку
checkBtn.addEventListener("click", runCheck);
