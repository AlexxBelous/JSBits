// 1. Назначаем обработчик клика на кнопку
document.getElementById("checkBtn").addEventListener("click", () => {

    // 2. Получаем случайный массив из DB
    const numbers = randomNumberSet();

    // 3. Выводим массив на страницу
    document.getElementById("array").textContent = numbers.join(", ");

    // 4. Проверяем, отсортирован ли массив
    let isSorted = true;

    for (let i = 0; i < numbers.length - 1; i++) {
        // 4.1. Если текущий элемент больше следующего — массив не отсортирован
        if (numbers[i] > numbers[i + 1]) {
            isSorted = false;
            break;
        }
    }

    // 5. Выводим результат проверки на страницу
    document.getElementById("result").textContent = isSorted
        ? "Array is sorted"
        : "Array is not sorted";
});
