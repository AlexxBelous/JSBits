// 1. Получаем элементы
const products = document.querySelectorAll(".product");
const resultEl = document.getElementById("result");

// 2. Функция парсинга диапазона цен
function parsePriceRange(rangeString) {
    const [min, max] = rangeString
        .split("-")
        .map(Number);

    return { min, max };
}

// 3. Функция расчёта средней цены
function calculateAverage(min, max) {
    return (min + max) / 2;
}

// 4. Рендер
function renderProductAverages(products) {
    products.forEach(product => {
        const { price } = product.dataset;

        const { min, max } = parsePriceRange(price);
        const average = calculateAverage(min, max);

        const line = document.createElement("div");
        line.textContent = `${product.textContent}: ${average}`;

        resultEl.appendChild(line);
    });
}

// 5. Запуск
renderProductAverages(products);
